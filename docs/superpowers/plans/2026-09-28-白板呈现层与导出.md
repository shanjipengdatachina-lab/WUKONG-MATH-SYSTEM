> **三个已核实的坑（不照做会红）：**
> 1. **全站 toast 是挂在 `document.body` 上的**（`shell.js` 的 `toastStack()`），不在页面外壳里，所以不能用 `.wb-shell .toast-stack` 去选它。**但也不用选** —— 把规则写进 `whiteboard.html` 自己的 `<style>` 里，它天生只在这一页生效。
> 2. **画笔配色面板是事件委托**（`#wb-colors` 上挂一个 click 监听、内部再 `closest`），所以换板时直接 `buildPickers()` 重建 `innerHTML` 是安全的，不会掉监听。
> 3. **`contentBounds()` 永远不空** —— 它写死了 `720 × 120` 的默认区域（为了让"适应窗口"总有合理范围）。导出判断"板上有没有东西"**不能用它**，要新写 `inkBounds()`。

```js
  /* 某个颜色在给定配色里是第几支；不在里面返回 -1 */
  function indexOfColor(list, value) {
    for (var i = 0; i < list.length; i++) { if (list[i].value === value) return i; }
    return -1;
  }

  /* 换板。
     笔色按**序号**平移 —— 唯一不会让人意外的映射（否则"我选的红色怎么变绿了"）。
     板上**已有的每一笔也一起平移**：不平移的话，切到黑板时原来的深色墨迹直接看不见，
     那不叫"换了块板"，那叫"把我的推导弄没了"。只动颜色，点坐标 / 笔粗 / 图形类型一律不动。 */
  function setTheme(key) {
    if (THEME_KEYS.indexOf(key) < 0) return false;
    if (key === state.theme) return true;
    var from = theme().colors;
    var to = THEMES[key].colors;
    state.theme = key;

    var ci = indexOfColor(from, state.color);
    if (ci >= 0 && to[ci]) state.color = to[ci].value;
    for (var i = 0; i < state.strokes.length; i++) {
      var s = state.strokes[i];
      var si = indexOfColor(from, s.color);
      if (si >= 0 && to[si]) s.color = to[si].value;
    }

    /* 当前这套笔整体换掉，并且刷新测试钩子上的快照（那是对象，不会自己跟着变） */
    COLORS = to;
    if (typeof window !== 'undefined' && window.__WB__) {
      window.__WB__.COLORS = to;
      if (window.__WB__.config) window.__WB__.config.COLORS = to;
    }
    /* 屏幕上的浮层（按钮 / 台阶框 / 答案块）**保持浅色** —— 它们是界面，不是板面。
       深色板上"深板 + 浅浮层"是常见做法（像暗色画布配浅色面板）。 */
    if (wrap && wrap.style && wrap.style.setProperty) wrap.style.setProperty('--wb-board', theme().board);

    buildPickers();
    syncUI();
    redraw();
    persist();
    return true;
  }
```

```js
  /* 板上的**真实**内容范围（画布单位）。
     和 contentBounds() 的区别：那个里面写死了 720×120 的默认区，为了"适应窗口"永远有合理范围，

  function inkBounds() {
    var minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
    var any = false;
    if (state.showProblem && problemBox.h > 0) {
      minX = problemBox.x; minY = problemBox.y;
      maxX = problemBox.x + problemBox.w; maxY = problemBox.y + problemBox.h;
      any = true;
    }
    for (var i = 0; i < state.strokes.length; i++) {
      var pts = state.strokes[i].points || [];
      for (var j = 0; j < pts.length; j++) {
        if (pts[j].x < minX) minX = pts[j].x;
        if (pts[j].y < minY) minY = pts[j].y;
        if (pts[j].x > maxX) maxX = pts[j].x;
        if (pts[j].y > maxY) maxY = pts[j].y;
        any = true;
      }
    }
    if (!any || !isFinite(minX)) return null;
    return { minX: minX, minY: minY, maxX: maxX, maxY: maxY };
  }
```

- [ ] **Step 4: 加导出几何**

在 `inkBounds()` 的**下方**插入：

```js
  /* ---------- 导出图片：先算几何，再画 ---------- */

  var EXPORT_LONG_EDGE = 1600;   // 输出长边上限
  var RATIOS = { '16:9': 16 / 9, '4:3': 4 / 3 };

  /* 纯函数：给定内容范围、比例、带不带分析/答案，算出整张图的布局。
     不改任何状态、不碰画布 —— 所以能直接断言（光栅化那一层很薄，在真机上验）。 */
  function exportPlan(bounds, ratio, include) {
    var b = bounds || { minX: 0, minY: 0, maxX: 400, maxY: 300 };
    var cw = Math.max(1, b.maxX - b.minX);
    var ch = Math.max(1, b.maxY - b.minY);
    var pad = Math.max(24, 0.08 * Math.max(cw, ch));
    var boardW = cw + 2 * pad;
    var boardH = ch + 2 * pad;

    var withPanel = !!(include && (include.analysis || include.answer));

    /* 一、整张图的输出尺寸。
       16:9 / 4:3 是**固定**的：1600×900 / 1600×1200 —— 内容装不下就补板面色。
       "按内容"才按内容比例算。 */
    var kFree = EXPORT_LONG_EDGE / Math.max(boardW, boardH);
    var freeW = Math.round(boardW * kFree);
    var freeH = Math.round(boardH * kFree);

    /* 二、右栏占多宽。先定它，板面区吃剩下的部分。
       注意：16:9 带右栏时**整张图还是 1600×900**，变窄的是板面区 ——
       这是"带上右栏板面就变小"的由来，不是把图整体变宽。 */
    var panel = null;
    var pw = 0, gap = 0;
    if (withPanel) {
      var refW = (ratio === 'free') ? freeW : EXPORT_LONG_EDGE;
      pw = Math.round(Math.min(620, Math.max(380, refW * 0.38)));
      gap = Math.round(refW * 0.02);
    }

    var outW, outH;
    if (ratio === 'free') {
      outW = freeW + pw + gap;
      outH = freeH;
    } else {
      var r = RATIOS[ratio] || RATIOS['16:9'];
      outW = EXPORT_LONG_EDGE;
      outH = Math.round(outW / r);
    }

    var boardArea = { x: 0, y: 0, w: outW - pw - gap, h: outH };
    if (pw) panel = { x: outW - pw, y: 0, w: pw, h: outH };

    /* 三、内容放进板面区：等比缩放 + 居中，多出来的边用板面色补 */
    var k = Math.min(boardArea.w / boardW, boardArea.h / boardH);
    var originX = boardArea.x + (boardArea.w - boardW * k) / 2 - (b.minX - pad) * k;
    var originY = boardArea.y + (boardArea.h - boardH * k) / 2 - (b.minY - pad) * k;

    return {
      out: { w: outW, h: outH },
      board: { x: boardArea.x, y: boardArea.y, w: boardArea.w, h: boardArea.h, scale: k, originX: originX, originY: originY },
      panel: panel
    };
  }

  /* 纯函数：右栏要写哪些字（与屏幕上的开合无关 —— 勾的是"这张图里要不要"） */
  function exportPanelLines(include) {
    var out = { steps: [], answer: null };
    if (!include) return out;
    if (include.analysis) {
      var gs = stepGuides();
      for (var i = 0; i < gs.length; i++) {
        out.steps.push({ no: i + 1, guide: gs[i], text: stepText(i) });
      }
    }
    if (include.answer) {
      var a = answerFor();
      out.answer = a ? { lines: a.lines, notes: !!state.analysis.showNotes } : { lines: [], notes: false, empty: true };
    }
    return out;
  }

  /* 纯函数：导出文件名。来源里的 · / 空格 / 斜杠都清成 -，避免文件名里有怪字符 */
  function exportFilename(data, dateMs) {
    var d = new Date(dateMs || Date.now());
    var ymd = d.getFullYear() + String(d.getMonth() + 1).padStart(2, '0') + String(d.getDate()).padStart(2, '0');
    var src = data && data.tag ? String(data.tag) : '';
    src = src.replace(/[·・/\\\s]+/g, '-').replace(/^-+|-+$/g, '');
    return '悟空数学-' + (src || '白板') + '-' + ymd + '.png';
  }
```

在 `assets/js/whiteboard.js` 里 `function exportFilename(...)` 的**下方**插入：

```js
  /* 状态：上次用的比例与勾选 */
  function shotState() {
    if (!state.shot) state.shot = { ratio: '16:9', include: { analysis: false, answer: false } };
    if (!state.shot.include) state.shot.include = { analysis: false, answer: false };
    return state.shot;
  }

  /* 离屏把板面画一遍。
     注意：**只画网格 / 题面 / 笔迹**，不画任何 UI。
     悬停底和把手必须显式关掉 —— 它们画在画布上，正悬停着导出就会跟着进去。 */
  function renderExport(plan) {
    if (!plan) return null;
    var oc = null;
    try { oc = document.createElement('canvas'); } catch (e) { oc = null; }
    if (!oc || typeof oc.getContext !== 'function') return null;
    var octx = oc.getContext('2d');
    if (!octx) return null;

    oc.width = plan.out.w;
    oc.height = plan.out.h;
    var th = theme();

    /* 先整张铺板面色：补边色自然就是板面色，也保证导出不带透明区 */
    octx.fillStyle = th.board;
    octx.fillRect(0, 0, plan.out.w, plan.out.h);

    /* 板面区：把原来那套"画世界坐标"的代码借过来用。
       做法是把模块级的 ctx 临时换成离屏的 —— 绘制函数都闭包在这个变量上，
       换掉就能复用整套渲染，不必把它们逐个改造成接收 ctx 参数。 */
    var real = ctx;
    var realHover = state.problemHover, realGrip = state.problemGripHot;
    state.problemHover = false; state.problemGripHot = false;
    ctx = octx;
    try {
      if (octx.setTransform) {
        octx.setTransform(plan.board.scale, 0, 0, plan.board.scale, plan.board.originX, plan.board.originY);
      }
      if (state.grid) drawGrid();
      drawProblemLayer();
      for (var i = 0; i < state.strokes.length; i++) drawStroke(state.strokes[i]);
    } finally {
      ctx = real;
      state.problemHover = realHover; state.problemGripHot = realGrip;
    }

    if (plan.panel) drawExportPanel(octx, plan.panel, exportPanelLines(shotState().include));
    return oc;
  }

  /* 右栏：用同一份数据**重新排版**，不是照抄屏幕上的浮层。
     理由是屏幕上的台阶框是可以随手拖的"工作台"，摆位是思考过程；
     交出去的图该是整齐的（决定 #13）。 */
  function drawExportPanel(octx, panel, data) {
    var th = theme();
    var pad = Math.round(panel.w * 0.08);
    var x = panel.x + pad;
    var maxW = panel.w - pad * 2;
    var y = panel.y + pad;

    /* 一条竖分隔线，把板面和右栏分开 */
    if (octx.setTransform) octx.setTransform(1, 0, 0, 1, 0, 0);
    octx.strokeStyle = th.grid;
    octx.lineWidth = 1;
    octx.beginPath();
    octx.moveTo(panel.x - 0.5, panel.y + pad);
    octx.lineTo(panel.x - 0.5, panel.y + panel.h - pad);
    octx.stroke();

    octx.textBaseline = 'top';
    /* 标题「分析」 */
    if (data.steps.length) {
      octx.font = '600 13px ' + UI_FONT;
      octx.fillStyle = th.tag;
      octx.fillText('分析', x, y);
      y += 22;
      for (var i = 0; i < data.steps.length; i++) {
        var st = data.steps[i];
        octx.font = '600 13px ' + MATH_FONT;
        octx.fillStyle = th.ink;
        octx.fillText(String(st.no), x, y + 1);
        octx.font = '14px ' + UI_FONT;
        octx.fillStyle = th.ink;
        var gLines = wrapText(st.guide, maxW - 22);
        for (var g = 0; g < gLines.length; g++) {
          octx.fillText(gLines[g], x + 22, y);
          y += 21;
        }
        if (st.text) {
          octx.font = '14px ' + MATH_FONT;
          octx.fillStyle = th.inkSoft;
          var tLines = wrapText(st.text, maxW - 22);
          for (var t = 0; t < tLines.length; t++) {
            octx.fillText(tLines[t], x + 22, y);
            y += 21;
          }
        }
        y += 8;
        if (y > panel.y + panel.h - pad) break;   /* 右栏装不下就停，别画到框外 */
      }
    }
    /* 标题「答案」 */
    if (data.answer) {
      if (y > panel.y + pad) y += 6;
      octx.font = '600 13px ' + UI_FONT;
      octx.fillStyle = th.tag;
      octx.fillText('答案', x, y);
      y += 22;
      var api = (typeof window !== 'undefined' && window.WK_ANALYSIS) ? window.WK_ANALYSIS : null;
      var lines = (data.answer.lines && data.answer.lines.length) ? data.answer.lines : null;
      if (!lines) {
        octx.font = '14px ' + UI_FONT;
        octx.fillStyle = th.inkSoft;
        var eLines = wrapText((api && api.NO_ANSWER) || '', maxW);
        for (var e = 0; e < eLines.length; e++) { octx.fillText(eLines[e], x, y); y += 21; }
      } else {
        for (var k = 0; k < lines.length; k++) {
          if (lines[k].text) {
            octx.font = '600 15px ' + MATH_FONT;
            octx.fillStyle = th.ink;
            var aLines = wrapText(lines[k].text, maxW);
            for (var a = 0; a < aLines.length; a++) { octx.fillText(aLines[a], x, y); y += 22; }
          }
          if (data.answer.notes && lines[k].note) {
            octx.font = '13.5px ' + UI_FONT;
            octx.fillStyle = th.inkSoft;
            var nLines = wrapText(lines[k].note, maxW);
            for (var n = 0; n < nLines.length; n++) { octx.fillText(nLines[n], x, y); y += 20; }
          }
          if (y > panel.y + panel.h - pad) break;
        }
      }
    }
  }

  /* 打开预览：先算几何、再画、再把图贴进 <img>。
     板上没内容就不做图。 */
  function openShot() {
    var b = inkBounds();
    if (!b) { toast('板上还没有内容'); return false; }
    shotState();
    var plan = exportPlan(b, state.shot.ratio, state.shot.include);
    var oc = renderExport(plan);
    if (!oc) { toast('这个环境不支持导出图片'); return false; }
    var img = byId('wb-shot-img');
    if (img && typeof oc.toDataURL === 'function') img.src = oc.toDataURL('image/png');
    syncShotUI();
    toggleHidden(byId('wb-shot-pop'), false);
    return true;
  }

  function closeShot() {
    toggleHidden(byId('wb-shot-pop'), true);
    var img = byId('wb-shot-img');
    if (img) img.src = '';
    return true;
  }

  function syncShotUI() {
    var s = shotState();
    var ca = byId('wb-shot-include-analysis');
    var cb = byId('wb-shot-include-answer');
    if (ca && 'checked' in ca) ca.checked = !!s.include.analysis;
    if (cb && 'checked' in cb) cb.checked = !!s.include.answer;
    var box = byId('wb-shot-ratios');
    if (box) {
      var keys = [['16:9', '16:9'], ['4:3', '4:3'], ['free', '按内容']];
      box.innerHTML = keys.map(function (kv) {
        return '<button type="button" class="wb-shot__ratio' + (s.ratio === kv[0] ? ' is-on' : '') +
          '" data-wb-ratio="' + kv[0] + '" aria-pressed="' + (s.ratio === kv[0] ? 'true' : 'false') + '">' +
          kv[1] + '</button>';
      }).join('');
    }
  }

  /* 重新算 + 重画 + 换图。任何勾选 / 比例变化都走这里 */
  function refreshShot() {
    var b = inkBounds();
    if (!b) return false;
    var plan = exportPlan(b, shotState().ratio, shotState().include);
    var oc = renderExport(plan);
    if (!oc) return false;
    var img = byId('wb-shot-img');
    if (img && typeof oc.toDataURL === 'function') img.src = oc.toDataURL('image/png');
    syncShotUI();
    return true;
  }

  function saveShot() {
    var img = byId('wb-shot-img');
    if (!img || !img.src) return false;
    var name = exportFilename(cardData(), Date.now());
    var a = document.createElement('a');
    if (!a || typeof a.click !== 'function') return false;
    a.href = img.src;
    a.download = name;
    a.click();
    toast('图片已保存：' + name);
    closeShot();
    return true;
  }
```
